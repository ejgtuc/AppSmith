SELECT string_agg(
         id::text
         || ' | codigo=[' || codigo || ']'
         || ' | tipo=[' || COALESCE(TRIM(tipo), '') || ']'
         || ' | desc=[' || COALESCE(descripcion, '') || ']'
         || ' | num=[' || COALESCE(numero::text, '') || ']'
         || ' | tip_grab=[' || COALESCE(TRIM(tip_grab), '') || ']',
         E'\n' ORDER BY id DESC) AS resumen
FROM (SELECT * FROM public.param ORDER BY id DESC LIMIT 5) t;
