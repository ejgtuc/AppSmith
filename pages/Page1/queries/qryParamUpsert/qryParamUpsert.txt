INSERT INTO public.param
  (codigo, descripcion, tipo, numero, caracter, fecha, logico, modulos, timestamp,
   fec_grab, usr_grab, wks_grab, tip_grab, prg_grab)
VALUES
  (NULLIF({{ inpCodigo.text || '' }}, ''),
   NULLIF({{ inpDescripcion.text || '' }}, ''),
   NULLIF({{ selTipo.selectedOptionValue || '' }}, ''),
   NULLIF(CASE WHEN {{ selTipo.selectedOptionValue || '' }} = 'N' THEN {{ inpNumero.text || '' }} ELSE '' END, '')::numeric,
   NULLIF(CASE WHEN {{ selTipo.selectedOptionValue || '' }} = 'C' THEN {{ inpCaracter.text || '' }} ELSE '' END, ''),
   NULLIF(CASE WHEN {{ selTipo.selectedOptionValue || '' }} = 'D' THEN {{ dpFecha.formattedDate || '' }} ELSE '' END, '')::date,
   NULLIF(CASE WHEN {{ selTipo.selectedOptionValue || '' }} = 'L' THEN {{ selLogico.selectedOptionValue || '' }} ELSE '' END, ''),
   NULLIF({{ inpModulos.text || '' }}, ''),
   NULLIF(CASE WHEN {{ selTipo.selectedOptionValue || '' }} = 'T' THEN {{ dpTimestamp.formattedDate || '' }} ELSE '' END, '')::timestamp,
   now(), 'WEB', 'LOCAL', 'A', 'appsmith/param')
ON CONFLICT (codigo) DO UPDATE SET
   descripcion = EXCLUDED.descripcion,
   tipo        = EXCLUDED.tipo,
   numero      = EXCLUDED.numero,
   caracter    = EXCLUDED.caracter,
   fecha       = EXCLUDED.fecha,
   logico      = EXCLUDED.logico,
   modulos     = EXCLUDED.modulos,
   timestamp   = EXCLUDED.timestamp,
   fec_grab    = now(),
   usr_grab    = 'WEB',
   wks_grab    = 'LOCAL',
   tip_grab    = 'M',
   prg_grab    = 'appsmith/param'
RETURNING id, (xmax = 0) AS insertado;
